function generateOrder() {

  const name = document.getElementById('name').value.trim() || '某人';

  const items = [];

  let totalCal = 0;

 

  // 1. 鸡蛋

  const egg = document.querySelector('input[name="egg"]:checked');

  if (egg) {

    items.push(`- 鸡蛋 ${egg.dataset.weight}：${egg.dataset.cal}大卡。${egg.dataset.desc}`);

    totalCal += parseInt(egg.dataset.cal);

  }

 

  // 2. 牛奶燕麦粥 & 坚果（通过 class="food" 统一处理）

  document.querySelectorAll('input.food:checked').forEach(cb => {

    items.push(`- ${cb.dataset.name} ${cb.dataset.weight}：${cb.dataset.cal}大卡。${cb.dataset.desc}`);

    totalCal += parseInt(cb.dataset.cal);

  });

 

  // 3. 蔬菜

  const veg = document.querySelector('input[name="veg"]:checked');

  if (veg) {

    items.push(`- ${veg.dataset.name} ${veg.dataset.weight}：${veg.dataset.cal}大卡。${veg.dataset.desc}`);

    totalCal += parseInt(veg.dataset.cal);

  }

 

  // 4. 碳水

  const carbs = document.querySelectorAll('input.carb:checked');

  if (carbs.length < 3) {

    alert("⚠️ 碳水类最少要选3项哦！请再选几项。");

    return;

  }

  if (carbs.length > 4) {

    if (!confirm("⚠️ 你选了5项碳水，热量和碳水会偏高，容易犯困。确定要全选吗？")) {

      return;

    }

  }

  carbs.forEach(cb => {

    items.push(`- ${cb.dataset.name} ${cb.dataset.weight}：${cb.dataset.cal}大卡。${cb.dataset.desc}`);

    totalCal += parseInt(cb.dataset.cal);

  });

 

  const text = `📢 早餐订单（${name}）\n${items.join('\n')}\n\n🔥 总热量约：${totalCal} 大卡\n\n💡 提醒：先吃蛋奶肉坚果，再吃蔬菜粗粮，最后吃主食；饭后漱口+牙线，30分钟后再刷牙。`;

 

  document.getElementById('resultText').value = text;

  document.getElementById('result').style.display = 'block';

  document.getElementById('result').scrollIntoView({ behavior: 'smooth' });

}

 

function copyResult() {

  const text = document.getElementById('resultText').value;

  if (navigator.clipboard && navigator.clipboard.writeText) {

    navigator.clipboard.writeText(text).then(() => {

      alert('已复制！请粘贴到微信群发回。');

    }).catch(() => {

      fallbackCopy();

    });

  } else {

    fallbackCopy();

  }

}

 

function fallbackCopy() {

  const textarea = document.getElementById('resultText');

  textarea.select();

  document.execCommand('copy');

  alert('已复制！请粘贴到微信群发回。');

}

 

function resetForm() {

  document.getElementById('breakfastForm').reset();

  document.getElementById('name').value = '';

  document.getElementById('result').style.display = 'none';

}
